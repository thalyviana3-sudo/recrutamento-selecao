create database if not exists Recrutamento_Seleção
character set utf8mb4
collate utf8mb4_unicode_ci;

Use recrutamento_seleção;


create table Area (
ID_Area int not null auto_increment primary key,
Nome_Area varchar (50) not null,
Nome_Diretor varchar (100) not null,
Telefone_Diretor varchar (20) not null,
Email_Diretor varchar (100) not null
);

create table Candidato (
ID_Candidato int not null auto_increment primary key,
Nome_Completo varchar (100) not null,
Data_Nascimento date not null,
CPF varchar (14) not null unique,
Sexo enum ('F','M') NOT null,
Email varchar (100) not null unique,
Telefone varchar (20) not null,
Escolaridade enum (
'Ensino Médio — Concluído',
'Ensino Superior — Cursando',
'Ensino Superior — Concluído',
'Pós-graduação — Cursando',
'Pós-graduação — Concluída',
'Outra Opção'
) NOT null,
Experiencia_Profissional varchar(500) not null,
Pretensao_Salarial decimal(10,2) not null
);

CREATE TABLE Departamento (
ID_Departamento INT NOT NULL AUTO_INCREMENT PRIMARY KEY,
Departamento VARCHAR(100) NOT NULL,
ID_Area INT NOT NULL,
FOREIGN KEY (ID_Area)
REFERENCES Area(ID_Area),
UNIQUE (Departamento, ID_Area)
);

create table Gestor(
ID_Gestor INT NOT NULL AUTO_INCREMENT PRIMARY KEY,
Nome_Gestor VARCHAR(100) NOT NULL,
Email_Gestor VARCHAR(100) NOT NULL UNIQUE,
Telefone_Gestor VARCHAR(20) NOT NULL,
ID_Departamento INT NOT NULL,
FOREIGN KEY (ID_Departamento)
REFERENCES Departamento(ID_Departamento)
);

CREATE TABLE Pacote_Beneficio (
ID_Pacote_Beneficio INT NOT NULL AUTO_INCREMENT PRIMARY KEY,
Nome_Pacote VARCHAR(100) NOT NULL UNIQUE,
Beneficios VARCHAR(500) NOT NULL
);

create table Vaga(
ID_Vaga int not null auto_increment primary key,
Data_Publicação date not null,
Qtd_Vaga int not null,
Titulo_Vaga varchar (100) not null,
ID_Departamento int not null,
ID_Gestor int not null,
Descricao varchar (500) not null,
Salario decimal (10,2) not null,
ID_Pacote_Beneficio int not null,
Contratacao enum (
'CLT',
'PJ'
) not null,
Horario_Trabalho enum(
'das 8H as 18H',
'das 9H as 19H'
) not null,
Modalidade enum (
'Presencial',
'Home Office'
) not null,
Nivel_Experiencia ENUM(
'Júnior',
'Pleno',
'Sênior'
) not null,
Escolaridade enum (
'Ensino Médio — Concluído',
'Ensino Superior — Cursando',
'Ensino Superior — Concluído',
'Pós-graduação — Cursando',
'Pós-graduação — Concluída',
'Outra Opção'
) not null,
Data_Fechamento date not null,
Status_Vaga enum (
'Aberta', 
'Em Andamento',
'Finalizada',
'Cancelada'
) not null,
foreign key (ID_Departamento)
references departamento(ID_Departamento),
foreign key (ID_Gestor)
references gestor(ID_Gestor),
foreign key (ID_Pacote_Beneficio)
references pacote_beneficio(ID_Pacote_Beneficio)
);

create table Processo_Seletivo(
ID_Processo int not null auto_increment primary key,
ID_Vaga int not null,
ID_Candidato int not null,
Data_Candidatura_Candidato date not null,
Fase_Processo_Seletivo enum(
'Candidatura',
'Triagem',
'Entrevista_RH',
'Entrevista_Gestor', 
'Proposta'
) not null,
Data_Entrevista_RH date,
Data_Entrevista_Gestor date,
Status_Candidato enum(
'Em Processo',
'Aprovado', 
'Reprovado', 
'Contratado'
) not null,
foreign key (ID_Candidato)
references candidato (ID_Candidato),
foreign key (ID_Vaga)
references vaga(ID_Vaga),
UNIQUE (ID_Vaga, ID_Candidato)
);

create table Endereco_Candidato(
ID_Endereço int not null auto_increment primary key,
ID_Candidato int not null,
Rua varchar (100) not null,
CEP varchar (9) not null,
Numero varchar (10) not null,
Estado varchar (50) not null,
Cidade varchar (50) not null,
Bairro varchar (50) not null,
Complemento varchar (100),
foreign key (ID_Candidato)
references candidato(ID_Candidato)
);
